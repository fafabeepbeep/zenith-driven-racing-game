//-----------------------------------------------------
//Global constant
//-----------------------------------------------------

//screen size
const SCREEN_W = 1920;
const SCREEN_H = 1080;

//coordinates of the screen centerpoint
const SCREEN_CX = SCREEN_W/2;
const SCREEN_CY = SCREEN_H/2;

//game states
const STATE_INIT = 1;
const STATE_RESTART = 2;
const STATE_PLAY = 3;
const STATE_GAMEOVER = 4;

//sprites
const PLAYER = 0;

//-----------------------------------------------------
//Global variables
//-----------------------------------------------------

 //current state
 var state = STATE_INIT;
//-----------------------------------------------------
//Main Scene
//-----------------------------------------------------

class MainScene extends Phaser.Scene
{
    constructor()
    {
        super({key: 'SceneMain'});
    }

/**loads all assets */
preload()
{
    this.load.image('imageBack', '../assets/img_back.png');
    this.load.image('imagePlayer', '../assets/img_player.png');
}

/**creates all objects */
create()
{
    //backgrounds
    this.sprBack = this.add.image(SCREEN_CX, SCREEN_CY,'imageBack' );

    //arrays of store sprites that will be manually drawn and used in the future on a rendering texture
    this.sprites = [this.add.image(0,0, 'imagePlayer').setVisible(false)]

    //instances
    this.circuit = new Circuit(this);
    this.camera = new Camera(this);
    this.player = new Player(this);
    this.settings = new Settings(this);

    this.socket = new WebSocket("ws://localhost:8765");

    this.socket.onopen = () =>
    {
        console.log("Connected to gesture server");
    };

    this.socket.onmessage = (event) =>
    {
        let data = JSON.parse(event.data);
        this.currentGesture = data.gesture;
    };

    this.socket.onerror = (err) =>
    {
        console.error("WebSocket error:", err);
    };

    this.socket.onclose = () =>
    {
        console.log("WebSocket closed");
    };

    //gesture indicator text
    this.gestureText = this.add.text(20, 20, "Gesture: NONE", {
        fontSize: "28px",
        fill: "#ffffff"
    });

    //camera ui overlay top left
    /*navigator.mediaDevices.getUserMedia({ video: true })
    .then(stream => {
    
        const video = document.createElement("video");
        video.srcObject = stream;
        video.play();
    
        video.style.position = "absolute";
        video.style.top = "10px";
        video.style.left = "10px";
        video.style.width = "240px";
        video.style.height = "180px";
        video.style.border = "2px solid white";
        video.style.zIndex = 1000;
    
        document.body.appendChild(video);
    });*/ 
    //this code make the browser access the camera instead of the python

    //listener pause game ------ this will be edited to suit the hand gesture system
    this.input.keyboard.on('keydown-P', function()
    {
        /** console.log("Game paused. Press [P] to resume.");
         * this is console ouput
         * instead of using console output
         * now we use the text object settings.js
        */
       
        this.settings.txtPause.text = "[P] Resume";
        this.scene.pause();
        this.scene.launch('ScenePause');
    }, this);

    //listener on resume event
    this.events.on('resume', function()
    {
        /**console.log('Game is resumed. Press [P] to pause.');
         * console ouput change to settings class
        */
       this.settings.show();
    }, this);

    //listener for left and right 
    this.keyLeft = this.input.keyboard.addKey(
        Phaser.Input.Keyboard.KeyCodes.LEFT
    );
    
    this.keyRight = this.input.keyboard.addKey(
        Phaser.Input.Keyboard.KeyCodes.RIGHT
    );

    //listener for forward, reverse, breal
    this.keyUp = this.input.keyboard.addKey(
        Phaser.Input.Keyboard.KeyCodes.UP
    );
    
    this.keyDown = this.input.keyboard.addKey(
        Phaser.Input.Keyboard.KeyCodes.DOWN
    );
    
    this.keySpace = this.input.keyboard.addKey(
        Phaser.Input.Keyboard.KeyCodes.SPACE
    );
}

/**main game loop */
update(time,delta)
{
    switch(state) 
    {
        case STATE_INIT:
            this.camera.init();
            this.player.init();
            state = STATE_RESTART;
            break;

        case STATE_RESTART:
            this.circuit.create();
            this.player.restart();
            state = STATE_PLAY;
            break;
        
        case STATE_PLAY:
            //duration of the time period
            var dt = Math.min(1, delta/1000);
            // keyboard input
            //this.player.moveLeft = this.keyLeft.isDown; keyboards input
            //this.player.moveRight = this.keyRight.isDown;
            //this.player.accelerate = this.keyUp.isDown;
            //this.player.brake = this.keySpace.isDown;
            //this.player.reverse = this.keyDown.isDown;

            this.player.moveLeft = (this.currentGesture === "LEFT");
            this.player.moveRight = (this.currentGesture === "RIGHT");
            this.player.accelerate = (this.currentGesture === "FORWARD");
            this.player.brake = (this.currentGesture === "STOP");
            this.player.reverse = false;
            this.player.update(dt);
            this.camera.update();
            this.circuit.render3D();
            //state = STATE_GAMEOVER;
            break;

        case STATE_GAMEOVER:
            break;
    }

    this.gestureText.setText("Gesture: " + this.currentGesture);
    console.log(
        this.keyLeft.isDown,
        this.keyRight.isDown
    );
}
}



//-----------------------------------------------------
//Main Scene
//-----------------------------------------------------

class PauseScene extends Phaser.Scene
{
    constructor()
    {
        super({key: 'ScenePause'});

        //websocket client
        this.socket = null;
        this.currentGesture = "NONE";
    }

    create()
    {
        //listener to resume game
        this.input.keyboard.on ('keydown-P', function()
        {
            this.scene.resume('SceneMain');
            this.scene.stop();
        }, this);
    }
}

//-----------------------------------------------------
//Initializing phaser game
//-----------------------------------------------------

//game configuration
var config =
{
    type: Phaser.AUTO,
    width: SCREEN_W,
    height: SCREEN_H,

    scale: 
    {
        mode: Phaser.Scale.FIT,
        autoCenter: Phaser.Scale.CENTER_BOTH,
    },

    scene: [MainScene, PauseScene]
};

//game instance
var game = new Phaser.Game(config);
